fn f() {
  values = [" x "
    .trim()]
  call(" c " /* comment
    ends */ .trim())
  call(" d " /* comment
    ends */)
  call(" y "
    .trim())
  value = Pair.Some(" z "
    .trim())
  if values
    .isEmpty() { false }
  match value
    .trim() { _ => true }
  for item in values
    .map(v => v) { call(item) }
  for {
    [values
      .map(v => v)]
    break
  }
  match xs {
    [n: Int where accepts(" x "
      .trim())
     .None] => n
  }
}
